\section{Introduction}

This document presents the formal task model used in our research on multi-mode mixed-criticality (MC) scheduling. The model extends traditional two-mode Adaptive Mixed-Criticality (AMC) scheduling to support multiple criticality levels, enabling graded degradation rather than binary normal/degraded transitions.

\textbf{Key analysis principle:} Safety certification is performed for only two modes:
\begin{enumerate}
    \item \textbf{Normal mode ($L_0$):} All tasks execute with their normal budgets $C_i^{\mathrm{lo}}$
    \item \textbf{HI-criticality mode:} HI tasks execute with budget $C_i^{\mathrm{hi}}$, LO tasks are abandoned
\end{enumerate}
The intermediate degradation levels ($L_1$ through $L_{k-2}$) use runtime-only execution budgets that do not affect safety invariants.

The model builds upon the foundational work presented in~\cite{9804600}.

\section{Task Model}

\subsection{Basic Task Properties}

A task set $\tau = \{\tau_1, \tau_2, \dots, \tau_n\}$ consists of $n$ independent, sporadically arriving tasks. Each task $\tau_i$ is characterized by:

\begin{table}[h]
\centering
\caption{Task Properties}
\begin{tabular}{lll}
\toprule
Symbol & Name & Description \\
\midrule
$T_i \in \N^+$ & Period & Minimum inter-arrival time (ticks) \\
$D_i \in \N^+$ & Deadline & Relative deadline ($D_i = T_i$ for implicit deadlines) \\
$C_i^{(0)} \in \N^+$ & Normal budget & Maximum execution in $L_0$ (all tasks run normally) \\
$C_i^{(k-1)} \in \N^+$ & HI execution budget & Maximum execution in $L_{k-1}$ (HI tasks only) \\
$\mathrm{BCET}_i \in \N^+$ & Best-case execution time & $\mathrm{BCET}_i \leq C_i^{(0)}$ \\
\bottomrule
\end{tabular}
\end{table}

The intermediate budgets $C_i^{(1)}, \dots, C_i^{(k-2)}$ are runtime-only values used for graded degradation.

\subsection{Criticality Assignment}

Each task is assigned a criticality level:
\[
\kappa_i \in \{\mathrm{HI}, \mathrm{LO}\}
\]

Let $\tau^{\mathrm{HI}} = \{\tau_i \in \tau \mid \kappa_i = \mathrm{HI}\}$ and $\tau^{\mathrm{LO}} = \{\tau_i \in \tau \mid \kappa_i = \mathrm{LO}\}$ denote the HI-criticality and LO-criticality subsets, respectively.

\subsection{Utilisation}

The utilisation of task $\tau_i$ under different criticality modes is:

\[
U_i^{\mathrm{lo}} = \frac{C_i^{(0)}}{T_i}, \quad
U_i^{\mathrm{hi}} = \begin{cases}
    \frac{C_i^{(k-1)}}{T_i} & \text{if } \kappa_i = \mathrm{HI} \\
    0 & \text{if } \kappa_i = \mathrm{LO}
\end{cases}
\]

The total LO-criticality utilisation is:
\[
U = \sum_{i=1}^n U_i^{\mathrm{lo}}
\]

The HI-criticality utilisation factor (CF) relates aggregate utilisations:
\[
U^{\mathrm{hi}} = \sum_{\tau_i \in \tau^{\mathrm{HI}}} U_i^{\mathrm{hi}}, \quad
\text{where } U^{\mathrm{hi}} = CP \cdot CF \cdot U
\]

Here $CP = |\tau^{\mathrm{HI}}| / n$ is the criticality proportion.

\section{Multi-Level Degradation Model}

\subsection{Degradation Levels and Execution Budgets}

We define $k \geq 2$ degradation levels, where each level $L_x$ has an associated execution budget
$C_i^{(x)}$ for each task $\tau_i$. The budgets satisfy:

\[
C_i^{(0)} = C_i^{\mathrm{lo}} \leq C_i^{(1)} \leq \dots \leq C_i^{(k-2)} \leq C_i^{(k-1)} = C_i^{\mathrm{hi}}
\]

\begin{table}[h]
\centering
\caption{Degradation Level Structure}
\begin{tabular}{cccc}
\toprule
Level & Name & Trigger Condition & LO Task Handling \\
\midrule
$L_0$ & Normal & None & All tasks run normally \\
$L_1$ & Degraded-1 & HI task reaches $R_1$ & Drop some LO tasks (lowest priority) \\
$L_2$ & Degraded-2 & HI task reaches $R_2$ & Drop more LO tasks \\
$\vdots$ & $\vdots$ & $\vdots$ & $\vdots$ \\
$L_{k-1}$ & Fully Degraded & HI task reaches $R_{\mathrm{trigger}}$ & Drop all LO tasks \\
\bottomrule
\end{tabular}
\end{table}

The degradation level transition trigger points satisfy:
\[
0 < R_1 \leq R_2 \leq \dots \leq R_{k-2} < R_{k-1} = R_{\mathrm{trigger}}
\]

\subsection{Analysis Modes vs. Runtime Budgets}

Only two modes are analyzed for safety certification:

\begin{itemize}
    \item \textbf{Normal mode ($L_0$):} All tasks execute with budget $C_i^{(0)} = C_i^{\mathrm{lo}}$
    \item \textbf{HI-criticality mode:} HI tasks execute with budget $C_i^{(k-1)} = C_i^{\mathrm{hi}}$, LO tasks are abandoned
\end{itemize}

The intermediate budgets $C_i^{(1)}, \dots, C_i^{(k-2)}$ are used at runtime for graded degradation.
At run-time, the response time $R_i$ (WCRT) is computed and monitored to determine when to transition
between degradation levels. The safety invariants are guaranteed by the standard two-mode analysis.

\subsection{Trigger Point Definition}

The AMC-RH trigger point $R_{\mathrm{trigger}}$ is defined as the response time of an HI-criticality task
in normal mode:

\[
R_{\mathrm{trigger}} = R_i^{\mathrm{lo}} = C_i^{(0)} + \sum_{\tau_j \in \mathrm{hp}(i)} \left\lceil \frac{R_i^{\mathrm{lo}}}{T_j} \right\rceil C_j^{(0)}
\]

When an HI-criticality job reaches execution time $R_{\mathrm{trigger}}$, the system must transition
to fully degraded mode ($L_{k-1}$) to preserve safety.

\subsection{Mode Transition Protocol}

The system transitions between degradation levels according to:

\begin{align*}
L_x \to L_{x+1} &\quad \text{when } \exists \tau_i \in \tau^{\mathrm{HI}}: \text{active job reaches } R_i \in L_x \\
L_x \to L_y &\quad y < x \text{ on exit (direct or cascade, per protocol design)}
\end{align*}

Each level $L_x$ specifies:
\begin{itemize}
    \item A drop policy $\pi_x$ determining which LO tasks are abandoned
    \item An exit condition $\xi_x$ determining when to leave $L_x$
\end{itemize}

\section{Job Execution Model}

\subsection{Execution Time Distribution}

Each job's execution time is drawn at release:

\[
E_i \sim
\begin{cases}
\mathrm{Uniform}(\mathrm{BCET}_i, C_i^{(0)}) & \text{if } \kappa_i = \mathrm{LO} \\
\mathrm{Uniform}(\mathrm{BCET}_i, C_i^{(0)}) & \text{if } \kappa_i = \mathrm{HI}, \text{normal mode} \\
\mathrm{Uniform}(C_i^{(0)}, C_i^{(k-1)}) & \text{if } \kappa_i = \mathrm{HI}, \text{HI behaviour}
\end{cases}
\]

HI-criticality behaviour occurs with probability $FP = 1/N$ per job.

\subsection{Execution Budget Enforcement}

During execution, jobs are constrained by their budgets:

\[
\text{Actual execution} = \min(E_i, B_i)
\]
where $B_i = C_i^{(0)}$ in normal mode and $B_i = C_i^{(k-1)}$ in degraded mode for HI tasks.

\section{Schedulability Analysis}

\subsection{Response Time Analysis (RTA)}

For fixed-priority preemptive scheduling, the response time of task $\tau_i$ is computed via fixed-point iteration:

\textbf{Normal Mode Response Time:}
\[
R_i^{\mathrm{lo}} = C_i^{(0)} + \sum_{\tau_j \in \mathrm{hp}(i)} \left\lceil \frac{R_i^{\mathrm{lo}}}{T_j} \right\rceil C_j^{(0)}
\]

where $\mathrm{hp}(i)$ is the set of higher-priority tasks.

\textbf{HI-Criticality Response Time:}
\[
R_i^{\mathrm{hi}} = C_i^{(k-1)} + \sum_{\tau_j \in \mathrm{hp}_{\mathrm{HI}}(i)} \left\lceil \frac{R_i^{\mathrm{hi}}}{T_j} \right\rceil C_j^{(k-1)}
+ \sum_{\tau_k \in \mathrm{hp}_{\mathrm{LO}}(i)} \left\lceil \frac{R_i^{\mathrm{lo}}}{T_k} \right\rceil C_k^{(0)}
\]

where $\mathrm{hp}_{\mathrm{HI}}(i)$ and $\mathrm{hp}_{\mathrm{LO}}(i)$ are higher-priority HI-criticality and LO-criticality tasks, respectively.

\subsection{AMC-rtb Schedulability Test}

A task set is AMC-rtb schedulable if:
\[
R_i^{\mathrm{hi}} \leq D_i \quad \forall \tau_i \in \tau^{\mathrm{HI}}
\]
and
\[
R_i^{\mathrm{lo}} \leq D_i \quad \forall \tau_i \in \tau
\]
